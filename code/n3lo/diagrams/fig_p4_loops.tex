%% fig_p4_loops.tex -- the O(p^4) pi N loop graphs (26)-(37) of the N3LO calculation (work/n3lo),
%% pseudovector pi NN coupling (as evaluated).  Double-column width (7.0 in).
%% Row 1: anomalous-magnetic-moment (Pauli, crossed circle) insertions, 1PI;  row 2: the same, 1PR;
%% row 3: graphs with one c_i pi pi N N vertex (diamond).  Crossed and time-reversed graphs are not drawn.
\documentclass[border=0pt]{standalone}
\input{diagstyle}
\begin{document}
\begin{tikzpicture}
\useasboundingbox (0,1.48) rectangle (17.8,5.52);
%% ---------------- row 1: Pauli insertions, 1PI --------------------------------
% (26) both photons on the nucleon in the loop, one of them Pauli
\begin{scope}[shift={(0.15,4.2)}]
  \draw[pion] (0.6,0) arc (180:0:0.65);
  \GamOver{(0.05,1.2)}{(1.0,0)}
  \GamOver{(1.5,0)}{(2.45,1.2)}
  \draw[nuc] (0,0) -- (2.5,0);
  \Vdot{(0.6,0)} \Vdot{(1.0,0)} \Vdot{(1.9,0)} \Vamm{(1.5,0)}
  \Glabel{26}
\end{scope}
% (27) photon on the pion, Pauli photon on the nucleon in the loop
\begin{scope}[shift={(3.15,4.2)}]
  \draw[pion] (0.6,0) arc (180:0:0.65);
  \draw[gam] (0.05,1.15) -- ($(1.25,0)+(140:0.65)$);
  \GamOver{(1.35,0)}{(2.45,1.2)}
  \draw[nuc] (0,0) -- (2.5,0);
  \Vdot{(0.6,0)} \Vdot{(1.9,0)} \Vdot{($(1.25,0)+(140:0.65)$)} \Vamm{(1.35,0)}
  \Glabel{27}
\end{scope}
% (28) Kroll-Ruderman vertex, Pauli photon on the nucleon in the loop
\begin{scope}[shift={(6.15,4.2)}]
  \draw[pion] (0.6,0) arc (180:0:0.65);
  \draw[gam] (0.05,1.15) -- (0.6,0);
  \GamOver{(1.3,0)}{(2.45,1.2)}
  \draw[nuc] (0,0) -- (2.5,0);
  \Vdot{(0.6,0)} \Vdot{(1.9,0)} \Vamm{(1.3,0)}
  \Glabel{28}
\end{scope}
%% ---------------- row 1 (cont.): Pauli insertions, 1PR -----------------------
% (29) vertex correction with the Pauli vertex inside the loop; Born (charge) vertex right
\begin{scope}[shift={(9.15,4.2)}]
  \draw[pion] (0.4,0) arc (180:0:0.6);
  \GamOver{(0.0,1.2)}{(1.0,0)}
  \draw[gam] (2.05,0) -- (2.45,1.2);
  \draw[nuc] (0,0) -- (2.5,0);
  \Vdot{(0.4,0)} \Vdot{(1.6,0)} \Vdot{(2.05,0)} \Vamm{(1.0,0)}
  \Glabel{29}
\end{scope}
% (30) O(p^3) vertex correction (photon on the pion); Pauli Born vertex right
\begin{scope}[shift={(12.15,4.2)}]
  \draw[pion] (0.4,0) arc (180:0:0.6);
  \draw[gam] (0.0,1.15) -- ($(1.0,0)+(135:0.6)$);
  \draw[gam] (2.05,0) -- (2.45,1.2);
  \draw[nuc] (0,0) -- (2.5,0);
  \Vdot{(0.4,0)} \Vdot{(1.6,0)} \Vdot{($(1.0,0)+(135:0.6)$)} \Vamm{(2.05,0)}
  \Glabel{30}
\end{scope}
% (31) self-energy between a charge and a Pauli Born vertex
\begin{scope}[shift={(15.15,4.2)}]
  \draw[pion] (0.7,0) arc (180:0:0.55);
  \draw[gam] (0.0,1.2) -- (0.32,0);
  \draw[gam] (2.18,0) -- (2.5,1.2);
  \draw[nuc] (0,0) -- (2.5,0);
  \Vdot{(0.32,0)} \Vdot{(0.7,0)} \Vdot{(1.8,0)} \Vamm{(2.18,0)}
  \Glabel{31}
\end{scope}
%% ---------------- row 2: c_i graphs -------------------------------------------
% (32) c_i pi pi N N vertex, both photons on the pion loop
\begin{scope}[shift={(0.15,2.1)}]

  \Glabel{32}
\end{scope}
% (33) c_i vertex, gamma gamma pi pi seagull on the pion loop
\begin{scope}[shift={(3.15,2.1)}]
  \draw[pion] (1.25,0.45) circle (0.45);
  \draw[gam] (0.05,1.25) -- (1.25,0.9);
  \draw[gam] (1.25,0.9) -- (2.45,1.25);

  \Glabel{33}
\end{scope}
% (34) gamma pi pi N N vertex (c_2, c_3), photon on the pion loop
\begin{scope}[shift={(6.15,2.1)}]
  \draw[pion] (1.25,0.45) circle (0.45);
  \draw[pion] (1.25,0.45) circle (0.45);
  \GamOver{(0.05,1.0)}{(1.25,0)}
  \draw[gam] ($(1.25,0.45)+(30:0.45)$) -- (2.45,1.15);
  \draw[nuc] (0,0) -- (2.5,0);
  \Vci{(1.25,0)} \Vdot{($(1.25,0.45)+(30:0.45)$)}

\begin{scope}[shift={(9.15,2.1)}]

  \GamOver{(1.25,0)}{(2.45,1.0)}
